cmake_minimum_required(VERSION 3.18)

project(LuaJIT)

set(LUA_COMMAND "" CACHE FILEPATH "Path to existing Lua interpreter")
option(LUAJIT_ENABLE_CLI "Enable building LuaJIT CLI" OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_EX ${CMAKE_BINARY_DIR}/lib/$<CONFIG>)

set(MAJVER 2)
set(MINVER 1)
set(ABIVER 5.1)
set(NODOTABIVER 51)

set(DASM ${CMAKE_SOURCE_DIR}/dynasm/dynasm.lua)

if(WIN32)
    set(LJCOMPILE "/O2 /W3 /D_CRT_SECURE_NO_DEPRECATE /D_CRT_SECURE_NO_WARNINGS /DLUAJIT_OS=LUAJIT_OS_WINDOWS")
    set(LJDYNBUILD "/D_CRT_STDIO_INLINE=__declspec(dllexport)__inline /DLUA_BUILD_AS_DLL")
    set(LJVM_MODE peobj)

    if(CMAKE_SYSTEM_PROCESSOR MATCHES "i386|i686|x86" OR CMAKE_VS_PLATFORM_NAME STREQUAL "Win32")
        set(LJCOMPILE "${LJCOMPILE} /arch:SSE2")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_x86.dasc)
        set(DASMFLAGS -D WIN -D JIT -D FFI -D ENDIAN_LE -D FPU)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x64|x86_64|AMD64" OR CMAKE_VS_PLATFORM_NAME STREQUAL "x64")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_x64.dasc)
        set(DASMFLAGS -D WIN -D JIT -D FFI -D ENDIAN_LE -D FPU -D P64)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "arm" OR CMAKE_VS_PLATFORM_NAME STREQUAL "ARM")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_arm.dasc)
        set(DASMFLAGS -D WIN -D JIT -D FFI)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64" OR CMAKE_VS_PLATFORM_NAME STREQUAL "ARM64")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_arm64.dasc)
        set(DASMFLAGS -D WIN -D JIT -D FFI -D P64)
    endif()
elseif(UNIX AND NOT APPLE)
    set(LJCOMPILE "-O2 -fomit-frame-pointer -fno-stack-protector -Wall -fPIC -D_FILE_OFFSET_BITS=64 -D_LARGEFILE_SOURCE -U_FORTIFY_SOURCE -DLUAJIT_UNWIND_EXTERNAL -DLUAJIT_OS=LUAJIT_OS_LINUX")
    set(LJDYNBUILD "-shared")
    set(LJVM_MODE elfasm)

    if(CMAKE_SYSTEM_PROCESSOR MATCHES "i386|i686")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_x86.dasc)
        set(LJCOMPILE "-march=i686 -msse -msse2 -mfpmath=sse ${LJCOMPILE}")
        set(DASMFLAGS -D ENDIAN_LE -D JIT -D FFI -D FPU -D HFABI)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x64|x86_64|AMD64")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_x64.dasc)
        set(DASMFLAGS -D ENDIAN_LE -D P64 -D JIT -D FFI -D FPU -D HFABI)
    endif()
elseif(APPLE)
    set(LJCOMPILE "-O2 -fomit-frame-pointer -fno-stack-protector -Wall -fPIC -D_FILE_OFFSET_BITS=64 -D_LARGEFILE_SOURCE -U_FORTIFY_SOURCE -DLUAJIT_UNWIND_EXTERNAL -DLUAJIT_OS=LUAJIT_OS_OSX")
    set(LJDYNBUILD "-shared")
    set(LJVM_MODE machasm)

    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64" OR CMAKE_OSX_ARCHITECTURES MATCHES "arm64")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_arm64.dasc)
        set(DASMFLAGS -D ENDIAN_LE -D P64 -D JIT -D FFI -D DUALNUM -D FPU -D HFABI)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x64|x86_64|AMD64" OR CMAKE_OSX_ARCHITECTURES MATCHES "x86_64")
        set(DASC ${CMAKE_SOURCE_DIR}/src/vm_x64.dasc)
        set(DASMFLAGS -D ENDIAN_LE -D P64 -D JIT -D FFI -D FPU -D HFABI)
    endif()
endif()

if(UNIX AND NOT APPLE)
    find_library(LIBM_LIBRARY NAMES m)
    find_library(LIBDL_LIBRARY NAMES dl)
endif()

set(CMAKE_C_FLAGS ${LJCOMPILE})
set(CMAKE_ASM_FLAGS ${CMAKE_C_FLAGS})

find_package(Git)

add_subdirectory(src/host)
add_subdirectory(src)
add_subdirectory(etc)
